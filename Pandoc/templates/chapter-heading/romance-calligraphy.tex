%% This part says to style the titles for "chapters"
% \titleformat{\chapter}[display]

\usepackage{calligra}
% this font is very oblique, so when using it, push it over to the left a little so that it lines up with content proceeding it better
\newcommand{\chapterheaderfont}{\hspace{-.5 mm}\calligra}
\def\chapterheaderfontname{Calligra}

% \bfseries is only on the title, since Calligra has no bold
\titleformat{\chapter}[display]{\filcenter}{\chapterheaderfont\Large\adforn{60}\hspace{.5 mm}\chaptertitlename\ \Numberstringnum{\thechapter}\hspace{3 mm}\adforn{32}}{0.5mm}{\bfseries\chapterheadingfont \huge}
