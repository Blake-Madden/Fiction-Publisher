% \titleformat{\chapter}[display]

% just used in colophon
\let\chapterheaderfont\primaryfont
\let\chapterheaderfontname\primaryfontname

%% "Watermark": a huge, very light chapter number behind/above the title
\titleformat{\chapter}[display]
  {\filcenter}%
  {\fontsize{130}{130}\selectfont\color{black!15}\chapternumber}%
  {-2.6cm}%
  {\huge\chapterheadingfont}
